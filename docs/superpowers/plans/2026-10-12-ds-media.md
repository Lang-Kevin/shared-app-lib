## Review Focus

1. **Very small screens and large font scale** (320dp wide, fontScale 1.3). The flanks are `weight(1f)`, so the shutter stays centred. Readings are `maxLines = 1` with `autoSize`, so the value shrinks instead of wrapping. The framing hint flips above the foot line when it would run into the HUD band (Task 3 tests this).
2. **Landscape.** The 20% scrim can be shorter than the shutter row, so legibility rests on the text shadow. Check that this is acceptable. The spotlight hole is recomputed when the container size changes (hr rotates without recreation, `configChanges` includes orientation).
3. **Long localized labels.** The DE phase text gets symmetric 48dp side padding and `maxLines = 2`. Flank labels ellipsize. The spotlight card height is measured, not guessed, and the card scrolls if it is taller than the area.
4. **Missing Best value.** `MediaReading(value = null)` shows the placeholder `–` with no unit. The flank keeps its weight, so the shutter does not shift (Task 1 tests this).
5. **Recording toggled rapidly.** Slots swap instantly with no animation overlap, so hidden controls are never tappable. Components hold no state. Debouncing double taps on the shutter is the app's job and is documented in KDoc.

package com.kevin.shared.ui.media

import androidx.compose.foundation.text.TextAutoSize
import androidx.compose.material3.MaterialTheme
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.graphics.Shadow
import androidx.compose.ui.text.TextStyle
import androidx.compose.ui.text.style.TextAlign
import androidx.compose.ui.text.style.TextOverflow
import com.kevin.shared.ui.theme.AppTheme

/** Shadow alpha for emphasized over-video text, e.g. the accent Best label (mockup 02: 0.9 instead of 0.75). */
internal const val EmphasizedShadowAlpha = 0.9f

internal fun mediaTextStyle(base: TextStyle, color: Color, shadow: Shadow): TextStyle =
    base.copy(color = color, shadow = shadow)

internal fun emphasizedShadow(shadow: Shadow): Shadow =
    shadow.copy(color = shadow.color.copy(alpha = EmphasizedShadowAlpha))

/**
 * Text drawn directly on camera or video (spec §4.1 "Over video"): the token text shadow,
 * no box, no pill. Use [emphasized] for accent-coloured text, which needs the stronger shadow.
 * Single shadow only, no halo.
 */
@Composable
fun MediaText(
    text: String,
    modifier: Modifier = Modifier,
    style: TextStyle = MaterialTheme.typography.bodyMedium,
    color: Color = AppTheme.colors.onMedia,
    emphasized: Boolean = false,
    textAlign: TextAlign = TextAlign.Unspecified,
    maxLines: Int = Int.MAX_VALUE,
    overflow: TextOverflow = TextOverflow.Ellipsis,
    autoSize: TextAutoSize? = null,
) {
    val tokenShadow = AppTheme.colors.mediaTextShadow
    val shadow = if (emphasized) emphasizedShadow(tokenShadow) else tokenShadow
    Text(
        text = text,
        modifier = modifier,
        style = mediaTextStyle(style, color, shadow),
        textAlign = textAlign,
        maxLines = maxLines,
        overflow = overflow,
        autoSize = autoSize,
    )
}
```

package com.kevin.shared.ui.media

import androidx.compose.foundation.layout.Arrangement
import androidx.compose.foundation.layout.Column
import androidx.compose.foundation.layout.Row
import androidx.compose.foundation.layout.Spacer
import androidx.compose.foundation.layout.width
import androidx.compose.foundation.text.TextAutoSize
import androidx.compose.material3.MaterialTheme
import androidx.compose.runtime.Composable
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.text.font.FontWeight
import com.kevin.shared.ui.theme.AppTheme

/** Shown when a reading has no value yet, e.g. no Best before the first jump. En dash. */
const val MediaReadingPlaceholder = "–"

internal const val MediaLabelAlpha = 0.85f
internal const val MediaUnitAlpha = 0.80f

internal data class ReadingText(val value: String, val unit: String?)

/** A missing or blank value shows [placeholder] and drops the unit ("– cm" reads like a value). */
internal fun readingText(value: String?, unit: String?, placeholder: String = MediaReadingPlaceholder): ReadingText =
    if (value.isNullOrBlank()) ReadingText(placeholder, null)
    else ReadingText(value, unit?.takeIf { it.isNotBlank() })

/**
 * A labelled number on video, e.g. Last / Best beside the shutter (spec §7.3 A2). No box.
 * [accentLabel] colours the label in the accent with the emphasized shadow (Best).
 * The value is one line and shrinks via autoSize instead of wrapping.
 * [label], [value] and [unit] come from the app's string resources and formatters.
 */
@Composable
fun MediaReading(
    label: String,
    value: String?,
    modifier: Modifier = Modifier,
    unit: String? = null,
    accentLabel: Boolean = false,
    horizontalAlignment: Alignment.Horizontal = Alignment.Start,
) {
    val reading = readingText(value, unit)
    val onMedia = AppTheme.colors.onMedia
    val typography = MaterialTheme.typography
    Column(modifier = modifier, horizontalAlignment = horizontalAlignment) {
        MediaText(
            text = label.uppercase(),
            style = typography.labelSmall,
            color = if (accentLabel) MaterialTheme.colorScheme.primary else onMedia.copy(alpha = MediaLabelAlpha),
            emphasized = accentLabel,
            maxLines = 1,
        )
        Row(horizontalArrangement = Arrangement.aligned(horizontalAlignment)) {
            MediaText(
                text = reading.value,
                modifier = Modifier
                    .weight(1f, fill = false)
                    .alignByBaseline(),
                style = typography.headlineLarge.copy(fontWeight = FontWeight.Light),
                maxLines = 1,
                autoSize = TextAutoSize.StepBased(
                    minFontSize = typography.titleMedium.fontSize,
                    maxFontSize = typography.headlineLarge.fontSize,
                ),
            )
            val unitText = reading.unit
            if (unitText != null) {
                Spacer(Modifier.width(AppTheme.spacing.xs))
                MediaText(
                    text = unitText,
                    modifier = Modifier.alignByBaseline(),
                    style = typography.bodySmall,
                    color = onMedia.copy(alpha = MediaUnitAlpha),
                    maxLines = 1,
                )
            }
        }
    }
}
```

import androidx.compose.ui.semantics.semantics
import androidx.compose.ui.unit.Dp
import androidx.compose.ui.unit.dp
import com.kevin.shared.ui.theme.AppTheme

/** Shutter geometry (mockup 02 ratios, jump's 72dp size kept). */
object ShutterDefaults {
    val Size: Dp = 72.dp
    val RingWidth: Dp = 4.dp
    val IdleDiscSize: Dp = 56.dp
    val StopSquareSize: Dp = 24.dp
    val ProgressWidth: Dp = 4.dp
    const val DisabledAlpha: Float = 0.4f
}

internal fun shutterGlyphSize(recording: Boolean): Dp =
    if (recording) ShutterDefaults.StopSquareSize else ShutterDefaults.IdleDiscSize

/** Arc sweep in degrees, or null when no arc is drawn (idle, or no progress, or NaN). */
internal fun shutterSweep(recording: Boolean, progress: Float?): Float? =
    if (!recording || progress == null || progress.isNaN()) null
    else 360f * progress.coerceIn(0f, 1f)

/**
 * Record / stop button on video. Idle: white disc in a white ring. Recording: small square in
 * the recording colour. [progress] (0..1) draws an arc over the ring for capped modes; it is
 * not a timer, so it is optional and off by default (§7.3 A2).
 *
 * [contentDescription] is the app's localized action, e.g. "Start recording" / "Stop recording".
 * The button holds no state: the app debounces double taps and decides what [onClick] does.
 */
@Composable
fun ShutterButton(
    recording: Boolean,
    onClick: () -> Unit,
    contentDescription: String,
    modifier: Modifier = Modifier,
    enabled: Boolean = true,
    progress: Float? = null,
) {
    val colors = AppTheme.colors
    val sweep = shutterSweep(recording, progress)
    Box(
        modifier = modifier
            .size(ShutterDefaults.Size)
            .alpha(if (enabled) 1f else ShutterDefaults.DisabledAlpha)
            .drawWithContent {
                drawContent()
                if (sweep != null) {
                    val stroke = ShutterDefaults.ProgressWidth.toPx()
                    val inset = stroke / 2f
                    drawArc(
                        color = colors.recording,
                        startAngle = -90f,
                        sweepAngle = sweep,
                        useCenter = false,

import androidx.compose.foundation.Canvas
import androidx.compose.foundation.layout.Box

import androidx.compose.ui.layout.Layout
import androidx.compose.ui.unit.Constraints
import androidx.compose.ui.unit.Dp
import androidx.compose.ui.unit.dp
import com.kevin.shared.ui.theme.AppTheme
import kotlin.math.roundToInt

object FramingGuideDefaults {
    const val IdleAlpha: Float = 1f
    const val RecordingAlpha: Float = 0.55f
    val FootLineWidth: Dp = 1.5.dp
    val BracketWidth: Dp = 2.dp
    val BracketArm: Dp = 24.dp

    /** Spec §7.3 A2: full strength while idle, dimmed to 55% while recording. */
    fun alpha(recording: Boolean): Float = if (recording) RecordingAlpha else IdleAlpha

    /** Final mockup: out of frame turns the guide to the error colour; otherwise onMedia. */
    @Composable
    fun color(outOfFrame: Boolean): Color =
        if (outOfFrame) MaterialTheme.colorScheme.error else AppTheme.colors.onMedia
}

internal data class FramingGeometry(
    val footY: Float,
    val roiLeft: Float,
    val roiRight: Float,
    val roiTop: Float,
    val arm: Float,
)

/** Fractions are clamped to 0..1. The bracket arm never exceeds half the ROI width or half its height. */
internal fun framingGeometry(
    width: Float,
    height: Float,
    footLineFraction: Float,
    roiTopFraction: Float,
    roiWidthFraction: Float,
    armPx: Float,
): FramingGeometry {
    val footY = height * footLineFraction.coerceIn(0f, 1f)
    val roiTop = height * roiTopFraction.coerceIn(0f, 1f)
    val roiWidth = width * roiWidthFraction.coerceIn(0f, 1f)
    val roiLeft = (width - roiWidth) / 2f
    val arm = minOf(armPx, roiWidth / 2f, ((footY - roiTop) / 2f).coerceAtLeast(0f))
    return FramingGeometry(footY, roiLeft, roiLeft + roiWidth, roiTop, arm)
}

/** Hint goes [gap] below the foot line; if it would cross [bottomLimit] it flips above the line. */
internal fun framingHintTop(footY: Float, gap: Float, hintHeight: Float, bottomLimit: Float): Float {
    val below = footY + gap
    return if (below + hintHeight <= bottomLimit) below
    else (footY - gap - hintHeight).coerceAtLeast(0f)
}

private fun DrawScope.drawFraming(g: FramingGeometry, color: Color, footStroke: Float, bracketStroke: Float) {
    drawLine(color, start = Offset(0f, g.footY), end = Offset(size.width, g.footY), strokeWidth = footStroke)
    if (g.arm <= 0f) return
    fun bracket(x: Float, y: Float, dx: Float, dy: Float) {
        drawLine(color, start = Offset(x, y), end = Offset(x + dx * g.arm, y), strokeWidth = bracketStroke, cap = StrokeCap.Square)
        drawLine(color, start = Offset(x, y), end = Offset(x, y + dy * g.arm), strokeWidth = bracketStroke, cap = StrokeCap.Square)
    }
    bracket(g.roiLeft, g.roiTop, 1f, 1f)
    bracket(g.roiRight, g.roiTop, -1f, 1f)
    bracket(g.roiLeft, g.footY, 1f, -1f)
    bracket(g.roiRight, g.footY, -1f, -1f)
}

/**
 * Foot line plus ROI corner brackets over the camera preview. Give it the preview's bounds
 * (e.g. `Modifier.matchParentSize()` inside the preview box).
 *
 * - [alpha]: use `FramingGuideDefaults.alpha(recording)`. It dims the lines only, never the hint.
 * - [color]: use `FramingGuideDefaults.color(outOfFrame)`.
 * - [hint]: optional over-video hint (e.g. a centred `MediaText`), placed under the foot line.
 *   It flips above the line if it would reach [bottomInset] from the bottom. Pass the HUD's
 *   bottom band height there (`CameraHud(onBottomBandHeight = …)`, converted to dp).
 */
@Composable
fun FramingGuide(
    footLineFraction: Float,
    roiTopFraction: Float,
    roiWidthFraction: Float,
    modifier: Modifier = Modifier,
    alpha: Float = FramingGuideDefaults.IdleAlpha,
    color: Color = AppTheme.colors.onMedia,
    bottomInset: Dp = 0.dp,
    hint: (@Composable () -> Unit)? = null,
) {
    val gap = AppTheme.spacing.sm
    val hintSidePadding = AppTheme.spacing.lg
    val guideColor = color.copy(alpha = color.alpha * alpha.coerceIn(0f, 1f))
    Layout(
        modifier = modifier,
        content = {
            Canvas(Modifier.fillMaxSize()) {
                val g = framingGeometry(
                    width = size.width,
                    height = size.height,
                    footLineFraction = footLineFraction,
                    roiTopFraction = roiTopFraction,
                    roiWidthFraction = roiWidthFraction,
                    armPx = FramingGuideDefaults.BracketArm.toPx(),
                )
                drawFraming(g, guideColor, FramingGuideDefaults.FootLineWidth.toPx(), FramingGuideDefaults.BracketWidth.toPx())
            }
            if (hint != null) {
                Box(Modifier.padding(horizontal = hintSidePadding), contentAlignment = Alignment.Center) { hint() }
            }
        },
    ) { measurables, constraints ->
        val width = if (constraints.hasBoundedWidth) constraints.maxWidth else constraints.minWidth
        val height = if (constraints.hasBoundedHeight) constraints.maxHeight else constraints.minHeight
        val guide = measurables[0].measure(Constraints.fixed(width, height))
        val hintPlaceable = measurables.getOrNull(1)?.measure(Constraints(maxWidth = width))
        val footY = height * footLineFraction.coerceIn(0f, 1f)
        val bottomLimit = (height - bottomInset.toPx()).coerceAtLeast(0f)
        val gapPx = gap.toPx()
        layout(width, height) {
            guide.place(0, 0)
            if (hintPlaceable != null) {
                val top = framingHintTop(footY, gapPx, hintPlaceable.height.toFloat(), bottomLimit)
                hintPlaceable.place((width - hintPlaceable.width) / 2, top.roundToInt())
            }
        }
    }
}

import androidx.compose.ui.text.style.TextAlign
import androidx.compose.ui.unit.Dp
import androidx.compose.ui.unit.dp
import com.kevin.shared.ui.theme.AppTheme

/** §7.3 A2: scrim over the bottom 20% only. */
internal const val HudScrimHeightFraction = 0.20f

/** §7.3 A2: scrim ≤ 25%. The token mediaScrim is 0.28 (§4.4), so the HUD clamps it. */
internal const val HudScrimMaxAlpha = 0.25f

internal val HudTouchTarget: Dp = 48.dp

internal fun hudScrimPeakAlpha(tokenAlpha: Float): Float = minOf(tokenAlpha, HudScrimMaxAlpha)

internal data class HudSlots(val idleControls: Boolean, val readings: Boolean)

/** Idle shows the camera controls; recording shows only the readings (§7.3 A2). Never both. */
internal fun hudSlots(recording: Boolean): HudSlots =
    HudSlots(idleControls = !recording, readings = recording)

/**
 * Camera chrome in the A2 layout (spec §7.3), drawn over a full-bleed preview.
 *
 * - Top: [navigation] at the start; [phase] centred, uppercase, on the video with a shadow (no pill).
 * - Bottom: start flank, shutter, end flank. The flanks share the width equally, so the shutter
 *   stays centred whatever the flanks contain (including an empty or placeholder reading).
 *   Idle shows [idleStart]/[idleEnd]; recording shows [recordingStart]/[recordingEnd]. The swap
 *   is instant, so a hidden control is never tappable mid-animation.
 * - Scrim: gradient to ≤25% over the bottom 20%, full-bleed. [contentInsets] apply to the
 *   controls, not to the scrim.
 * - [onBottomBandHeight] reports the bottom band's height in px (including insets) so the app can
 *   keep the foot line and the framing hint clear of it (`FramingGuide(bottomInset = …)`).
 *
 * Stateless. The app owns recording state, phase strings (localized) and shutter debouncing.
 */
@Composable
fun CameraHud(
    recording: Boolean,
    modifier: Modifier = Modifier,
    phase: String? = null,
    phaseStyle: TextStyle = MaterialTheme.typography.titleMedium,
    contentInsets: WindowInsets = WindowInsets.safeDrawing,
    onBottomBandHeight: (Int) -> Unit = {},
    navigation: @Composable () -> Unit = {},
    idleStart: @Composable () -> Unit = {},
    idleEnd: @Composable () -> Unit = {},
    recordingStart: @Composable () -> Unit = {},
    recordingEnd: @Composable () -> Unit = {},
    shutter: @Composable () -> Unit,
) {
    val colors = AppTheme.colors
    val spacing = AppTheme.spacing
    val slots = hudSlots(recording)
    val scrimPeak = colors.mediaScrim.copy(alpha = hudScrimPeakAlpha(colors.mediaScrim.alpha))
    Box(modifier.fillMaxSize()) {
        Box(
            Modifier
                .align(Alignment.BottomCenter)
                .fillMaxWidth()
                .fillMaxHeight(HudScrimHeightFraction)
                .background(Brush.verticalGradient(listOf(Color.Transparent, scrimPeak))),
        )
        Box(
            Modifier
                .align(Alignment.TopCenter)
                .fillMaxWidth()
                .windowInsetsPadding(contentInsets.only(WindowInsetsSides.Top + WindowInsetsSides.Horizontal))
                .padding(spacing.sm)
                .heightIn(min = HudTouchTarget),
        ) {
            Box(Modifier.align(Alignment.CenterStart)) { navigation() }
            if (phase != null) {
                MediaText(
                    text = phase.uppercase(),
                    modifier = Modifier
                        .align(Alignment.Center)
                        .padding(horizontal = HudTouchTarget),
                    style = phaseStyle,
                    textAlign = TextAlign.Center,
                    maxLines = 2,
                )
            }
        }
        Row(
            Modifier
                .align(Alignment.BottomCenter)
                .fillMaxWidth()
                .onSizeChanged { onBottomBandHeight(it.height) }
                .windowInsetsPadding(contentInsets.only(WindowInsetsSides.Bottom + WindowInsetsSides.Horizontal))
                .padding(horizontal = spacing.lg, vertical = spacing.xl),
            verticalAlignment = Alignment.CenterVertically,
        ) {
            Box(Modifier.weight(1f), contentAlignment = Alignment.CenterStart) {
                if (slots.idleControls) idleStart()
                if (slots.readings) recordingStart()
            }
            Box(Modifier.padding(horizontal = spacing.md)) { shutter() }
            Box(Modifier.weight(1f), contentAlignment = Alignment.CenterEnd) {
                if (slots.idleControls) idleEnd()
                if (slots.readings) recordingEnd()
            }
        }
    }
}

/** Tracks the root bounds of tagged elements so the spotlight can cut a hole around them (from hr). */
class TutorialAnchors {
    private val rects = mutableStateMapOf<String, Rect>()
    private val requesters = mutableMapOf<String, BringIntoViewRequester>()

    @OptIn(ExperimentalFoundationApi::class)
    fun modifierFor(key: String): Modifier = Modifier
        .onGloballyPositioned { coords -> rects[key] = coords.boundsInRoot() }
        .bringIntoViewRequester(requesterFor(key))

    fun rectOf(key: String): Rect? = rects[key]

    @OptIn(ExperimentalFoundationApi::class)
    fun requesterFor(key: String): BringIntoViewRequester =
        requesters.getOrPut(key) { BringIntoViewRequester() }
}

@Composable
fun rememberTutorialAnchors(): TutorialAnchors = remember { TutorialAnchors() }

fun Modifier.tutorialAnchor(anchors: TutorialAnchors, key: String): Modifier =
    this.then(anchors.modifierFor(key))

/** Hole corner = MaterialTheme.shapes.medium (14dp); a Dp is needed for drawRoundRect. */
private val SpotlightCorner: Dp = 14.dp
private val SpotlightCardMaxWidth: Dp = 480.dp

/** Root coordinates → the overlay's own coordinates (the overlay may not start at the root origin). */
internal fun toOverlaySpace(rectInRoot: Rect, overlayOriginInRoot: Offset): Rect =
    rectInRoot.translate(-overlayOriginInRoot)

/** Target padded by [padPx], clamped to [bounds]; null when nothing of it is on screen. */
internal fun spotlightHole(target: Rect, padPx: Float, bounds: Size): Rect? {
    val hole = Rect(
        left = (target.left - padPx).coerceAtLeast(0f),
        top = (target.top - padPx).coerceAtLeast(0f),
        right = (target.right + padPx).coerceAtMost(bounds.width),
        bottom = (target.bottom + padPx).coerceAtMost(bounds.height),
    )
    return if (hole.width <= 0f || hole.height <= 0f) null else hole
}

/** Below the hole if the card fits there, else above if it fits, else the roomier side. No hole: bottom. */
internal fun spotlightCardAtTop(hole: Rect?, containerHeight: Float, cardHeight: Float, marginPx: Float): Boolean {
    if (hole == null) return false
    val needed = cardHeight + 2f * marginPx
    val spaceBelow = containerHeight - hole.bottom
    val spaceAbove = hole.top
    return when {
        spaceBelow >= needed -> false
        spaceAbove >= needed -> true
        else -> spaceAbove > spaceBelow
    }
}

internal fun nextStepIndex(index: Int, stepCount: Int): Int? =
    if (index + 1 < stepCount) index + 1 else null

/**
 * Dims the screen ([AppTheme.colors.scrimStrong]), cuts a hole around the current step's anchor and
 * shows an explanation card on the side with room. Blocks all touches except the card's buttons.
 * Place it last in a full-screen Box above the content. Seen-state persistence stays in the app.
 * All labels are the app's localized strings; [nextLabel] is composable (so it can call stringResource)
 * and receives the 1-based step and the total.
 */
@Composable
fun TutorialSpotlight(
    steps: List<TutorialStep>,
    anchors: TutorialAnchors,
    visible: Boolean,
    onFinish: () -> Unit,
    skipLabel: String,
    nextLabel: @Composable (step: Int, total: Int) -> String,
    doneLabel: String,
    modifier: Modifier = Modifier,
    contentInsets: WindowInsets = WindowInsets.safeDrawing,
) {
    if (!visible || steps.isEmpty()) return
    var index by remember(steps.size) { mutableIntStateOf(0) }
    val current = index.coerceIn(0, steps.lastIndex)
    val step = steps[current]

    var overlayOrigin by remember { mutableStateOf(Offset.Zero) }
    var areaOrigin by remember { mutableStateOf(Offset.Zero) }
    var containerSize by remember { mutableStateOf(IntSize.Zero) }
    var frozenKey by remember { mutableStateOf<String?>(null) }
    var frozenRect by remember { mutableStateOf<Rect?>(null) }

    LaunchedEffect(step.key, containerSize) {
        anchors.requesterFor(step.key).bringIntoView()
        withFrameNanos { } // one frame for layout to settle after scrolling (hr TutorialOverlay.kt:93)
        frozenRect = anchors.rectOf(step.key) // null if the target isn't composed: dim only, card at the bottom
        frozenKey = step.key
    }

    val spacing = AppTheme.spacing
    val scrim = AppTheme.colors.scrimStrong
    val padPx = with(LocalDensity.current) { spacing.sm.toPx() }
    val margin = spacing.lg
    val holeInOverlay = frozenRect
        ?.takeIf { frozenKey == step.key }
        ?.let { spotlightHole(toOverlaySpace(it, overlayOrigin), padPx, containerSize.toSize()) }
    val holeInArea = holeInOverlay?.translate(overlayOrigin - areaOrigin)
    val isLast = nextStepIndex(current, steps.size) == null

    Box(
        modifier
            .fillMaxSize()
            .onGloballyPositioned {
                overlayOrigin = it.positionInRoot()
                containerSize = it.size
            },
    ) {
        Canvas(
            Modifier
                .matchParentSize()
                .graphicsLayer { compositingStrategy = CompositingStrategy.Offscreen }
                .pointerInput(Unit) {
                    awaitPointerEventScope {
                        while (true) awaitPointerEvent().changes.forEach { it.consume() }
                    }
                },
        ) {
            drawRect(scrim)
            holeInOverlay?.let {
                drawRoundRect(
                    color = Color.Transparent,
                    topLeft = it.topLeft,
                    size = it.size,
                    cornerRadius = CornerRadius(SpotlightCorner.toPx()),
                    blendMode = BlendMode.Clear,
                )
            }
        }
        Layout(
            modifier = Modifier
                .fillMaxSize()
                .windowInsetsPadding(contentInsets)
                .onGloballyPositioned { areaOrigin = it.positionInRoot() },
            content = {
                SpotlightCard(
                    step = step,
                    actionLabel = if (isLast) doneLabel else nextLabel(current + 1, steps.size),
                    skipLabel = skipLabel,
                    onSkip = onFinish,
                    onAction = {
                        val next = nextStepIndex(current, steps.size)
                        if (next == null) onFinish() else index = next
                    },
                )
            },
        ) { measurables, constraints ->
            val marginPx = margin.roundToPx()
            val width = constraints.maxWidth
            val height = constraints.maxHeight
            val card = measurables.first().measure(
                Constraints(
                    maxWidth = (width - 2 * marginPx).coerceAtLeast(0),
                    maxHeight = (height - 2 * marginPx).coerceAtLeast(0),
                ),
            )
            val atTop = spotlightCardAtTop(holeInArea, height.toFloat(), card.height.toFloat(), marginPx.toFloat())
            layout(width, height) {
                card.place((width - card.width) / 2, if (atTop) marginPx else height - marginPx - card.height)
            }
        }
    }
}

@Composable

            .clip(shape)

            .border(1.dp, MaterialTheme.colorScheme.outline, shape)
            .verticalScroll(rememberScrollState())
            .padding(spacing.lg),

    ) {

        Row(
            Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.SpaceBetween,
            verticalAlignment = Alignment.CenterVertically,
        ) {
            TextButton(onClick = onSkip) {
                Text(skipLabel, color = MaterialTheme.colorScheme.onSurfaceVariant)
            }
            AccentPillButton(text = actionLabel, onClick = onAction)
        }
    }
}

enum class PermissionGateMode { Granted, Request, OpenSettings }

/**
 * Which action the gate offers.
 *
 * `shouldShowRationale` is false both before the first request and after a permanent denial,
 * so [requestedOnce] tells them apart. Keep it in `rememberSaveable` and set it to true in the
 * permission result callback. After an app restart it is false again: the first tap re-requests,
 * the system denies at once, the callback sets it, and the gate switches to OpenSettings.
 */
fun permissionGateMode(granted: Boolean, shouldShowRationale: Boolean, requestedOnce: Boolean): PermissionGateMode =
    when {
        granted -> PermissionGateMode.Granted
        shouldShowRationale -> PermissionGateMode.Request
        requestedOnce -> PermissionGateMode.OpenSettings
        else -> PermissionGateMode.Request
    }

/** Opens this app's system settings page, for permissions the user denied permanently. */
fun openAppSettings(context: Context) {
    val intent = Intent(Settings.ACTION_APPLICATION_DETAILS_SETTINGS, Uri.fromParts("package", context.packageName, null))
        .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
    context.startActivity(intent)
}

private val PermissionIconSize: Dp = 48.dp

/**
 * Full-screen gate shown until [granted]; then it renders [content]. Stateless: the app owns the
 * permission state and picks [actionLabel] and [onAction] from [permissionGateMode]. Example
 * (accompanist in the app):
 *
 *     var requestedOnce by rememberSaveable { mutableStateOf(false) }
 *     val perms = rememberMultiplePermissionsState(listOf(CAMERA, RECORD_AUDIO)) { requestedOnce = true }
 *     val mode = permissionGateMode(perms.allPermissionsGranted, perms.shouldShowRationale, requestedOnce)
 *     PermissionGate(
 *         granted = mode == PermissionGateMode.Granted,
 *         title = stringResource(R.string.camera_permission_title),
 *         message = stringResource(R.string.camera_permission_message),
 *         actionLabel = stringResource(if (mode == PermissionGateMode.OpenSettings) R.string.open_settings else R.string.grant),
 *         onAction = { if (mode == PermissionGateMode.OpenSettings) openAppSettings(context) else perms.launchMultiplePermissionRequest() },
 *     ) { RecordContent() }
 */
@Composable
fun PermissionGate(
    granted: Boolean,
    title: String,
    message: String,
    actionLabel: String,
    onAction: () -> Unit,
    modifier: Modifier = Modifier,
    icon: ImageVector? = null,
    content: @Composable () -> Unit,
) {
    if (granted) {
        content()
    } else {
        Box(
            modifier
                .fillMaxSize()
                .background(MaterialTheme.colorScheme.background)
                .windowInsetsPadding(WindowInsets.safeDrawing),
            contentAlignment = Alignment.Center,
        ) {
            Column(
                Modifier
                    .verticalScroll(rememberScrollState())
                    .padding(AppTheme.spacing.xl),
                verticalArrangement = Arrangement.spacedBy(AppTheme.spacing.lg),
                horizontalAlignment = Alignment.CenterHorizontally,
            ) {
                if (icon != null) {
                    Icon(
                        icon,
                        contentDescription = null,
                        tint = MaterialTheme.colorScheme.onSurfaceVariant,
                        modifier = Modifier.size(PermissionIconSize),
                    )
                }
                Text(
                    title.uppercase(),
                    style = MaterialTheme.typography.titleMedium,
                    color = MaterialTheme.colorScheme.onSurface,
                    textAlign = TextAlign.Center,
