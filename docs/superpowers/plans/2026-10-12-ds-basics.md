```kotlin
package com.kevin.shared.format

/** How [formatElapsed] renders a duration. No default on purpose: every call site picks one. */
enum class DurationStyle {
    /** Always `HH:MM:SS`, zero-padded (live timer, notification). */
    Hms,
    /** `MM:SS` under one hour, else `HH:MM:SS` (session lists, chart time). */
    Compact,
    /** `m:ss` under one hour, else `h:mm:ss`; the leading unit is unpadded (axis ticks, short clocks). */
    Clock,
}

/** Formats [seconds] of elapsed time. Negative input counts as 0. ASCII digits in every locale. */
fun formatElapsed(seconds: Long, style: DurationStyle): String {
    val total = seconds.coerceAtLeast(0)
    val h = total / 3600
    val m = (total % 3600) / 60
    val s = total % 60
    return when (style) {
        DurationStyle.Hms -> "${pad2(h)}:${pad2(m)}:${pad2(s)}"
        DurationStyle.Compact -> if (h > 0) "${pad2(h)}:${pad2(m)}:${pad2(s)}" else "${pad2(m)}:${pad2(s)}"
        DurationStyle.Clock -> if (h > 0) "$h:${pad2(m)}:${pad2(s)}" else "$m:${pad2(s)}"
    }
}

private fun pad2(value: Long): String = value.toString().padStart(2, '0')
```

```kotlin
package com.kevin.shared.format

import java.math.BigDecimal
import java.math.RoundingMode
import java.text.DecimalFormat
import java.text.DecimalFormatSymbols
import java.util.Locale

/** Shown instead of a number that doesn't exist (NaN, infinite, not measured). */
const val NO_VALUE: String = "—"

/**
 * Rounds HALF_UP to [maxDecimals] on the shortest decimal form of [value] (2.675 -> "2.68", like "%.2f"),
 * keeps at least [minDecimals], never groups thousands and takes the decimal separator from [locale].
 * NaN/infinite -> [NO_VALUE]. A result that rounds to zero has no minus sign.
 */
fun formatDecimal(value: Double, maxDecimals: Int, locale: Locale, minDecimals: Int = 0): String {
    require(maxDecimals >= 0) { "maxDecimals must be >= 0" }
    require(minDecimals in 0..maxDecimals) { "minDecimals must be in 0..maxDecimals" }
    if (!value.isFinite()) return NO_VALUE
    val rounded = BigDecimal.valueOf(value).setScale(maxDecimals, RoundingMode.HALF_UP)
    val format = DecimalFormat("0", DecimalFormatSymbols.getInstance(locale)).apply {
        minimumFractionDigits = minDecimals
        maximumFractionDigits = maxDecimals
        isGroupingUsed = false
    }
    return format.format(if (rounded.signum() == 0) BigDecimal.ZERO else rounded)
}

/** "[number] [unit]" joined by a no-break space (never wraps), e.g. "42.5 cm". Non-finite -> [NO_VALUE] alone. */
fun formatMeasure(value: Double, unit: String, maxDecimals: Int, locale: Locale, minDecimals: Int = 0): String {
    val number = formatDecimal(value, maxDecimals, locale, minDecimals)
    return if (number == NO_VALUE) NO_VALUE else "$number $unit"
}

/** [fraction] in 0..1 (clamped, NaN -> 0) as a whole percent, e.g. 0.25 -> "25%". */
fun formatPercent(fraction: Float, locale: Locale): String {
    val clamped = if (fraction.isNaN()) 0f else fraction.coerceIn(0f, 1f)
    return formatDecimal(clamped * 100.0, 0, locale) + "%"
}
```

```kotlin
import java.time.LocalDate
import java.time.ZoneId
import java.time.ZoneOffset
```

```kotlin
import java.util.Locale

/** Fixed numeric layouts shared by both apps; only the weekday abbreviation follows the locale. */
enum class DateStyle(internal val pattern: String, internal val hasTime: Boolean) {
    DayMonth("dd.MM", false),              // "12.10" (week labels)
    ShortDate("dd.MM.yy", false),          // "12.10.26"
    Date("dd.MM.yyyy", false),             // "12.10.2026"
    WeekdayDate("EEE, dd.MM.yyyy", false), // "Mon, 12.10.2026"
    WeekdayTime("EEE HH:mm", true),        // "Mon 18:42"
    Time("HH:mm", true),                   // "18:42"
    DateTime("dd.MM.yyyy HH:mm", true),    // "12.10.2026 18:42"
}

fun formatDate(epochMillis: Long, style: DateStyle, locale: Locale, zone: ZoneId): String =
    DateTimeFormatter.ofPattern(style.pattern, locale).format(Instant.ofEpochMilli(epochMillis).atZone(zone))

/** Calendar-day overload; [style] must not contain a time. */
fun formatDate(date: LocalDate, style: DateStyle, locale: Locale): String {
    require(!style.hasTime) { "$style needs a time; use the epochMillis overload" }
    return DateTimeFormatter.ofPattern(style.pattern, locale).format(date)
}

/** "start – end" (en dash), e.g. for a date-range filter chip. */
fun formatDateRange(start: LocalDate, end: LocalDate, style: DateStyle, locale: Locale): String =
    "${formatDate(start, style, locale)} – ${formatDate(end, style, locale)}"

/** M3 DatePicker/DateRangePicker millis are UTC midnight of the picked calendar day. */
fun pickerMillisToLocalDate(utcMillis: Long): LocalDate =
    Instant.ofEpochMilli(utcMillis).atZone(ZoneOffset.UTC).toLocalDate()

/** Inverse of [pickerMillisToLocalDate], for pre-selecting a day in the picker. */
fun LocalDate.toPickerMillis(): Long = atStartOfDay(ZoneOffset.UTC).toInstant().toEpochMilli()

/** Local midnight of this day in [zone], as epoch millis (start of a filter window). */
fun LocalDate.startOfDayMillis(zone: ZoneId): Long = atStartOfDay(zone).toInstant().toEpochMilli()
```

```kotlin
package com.kevin.shared.text

import android.content.Context
import android.content.res.Resources
import androidx.annotation.PluralsRes
import androidx.annotation.StringRes
import androidx.compose.runtime.Composable
import androidx.compose.ui.res.pluralStringResource
import androidx.compose.ui.res.stringResource

/** User-visible text a ViewModel can hold without a Context. Args may themselves be [UiText]. */
sealed interface UiText {
    data class Res(@StringRes val resId: Int, val args: List<Any> = emptyList()) : UiText
    data class Plural(@PluralsRes val resId: Int, val count: Int, val args: List<Any> = listOf(count)) : UiText
    data class Raw(val value: String) : UiText

    companion object {
        /** Same call shape as jump's former `data class UiText(resId, args)`: migration is an import swap. */
        operator fun invoke(@StringRes resId: Int, args: List<Any> = emptyList()): UiText.Res =
            UiText.Res(resId, args)
    }
}

/** Resource access behind [resolve]: [Resources] in production, a fake in JVM tests. */
interface StringLookup {
    fun string(@StringRes resId: Int, args: Array<out Any>): String
    fun plural(@PluralsRes resId: Int, count: Int, args: Array<out Any>): String
}

fun UiText.resolve(lookup: StringLookup): String = when (this) {
    is UiText.Raw -> value
    is UiText.Res -> lookup.string(resId, args.map { it.resolvedWith(lookup) }.toTypedArray())
    is UiText.Plural -> lookup.plural(resId, count, args.map { it.resolvedWith(lookup) }.toTypedArray())
}

private fun Any.resolvedWith(lookup: StringLookup): Any = if (this is UiText) resolve(lookup) else this

fun UiText.resolve(context: Context): String = resolve(ResourcesLookup(context.resources))

private class ResourcesLookup(private val resources: Resources) : StringLookup {
    override fun string(resId: Int, args: Array<out Any>): String = resources.getString(resId, *args)
    override fun plural(resId: Int, count: Int, args: Array<out Any>): String =
        resources.getQuantityString(resId, count, *args)
}

@Composable
fun UiText.resolve(): String = when (this) {
    is UiText.Raw -> value
    is UiText.Res -> stringResource(resId, *composeArgs(args))
    is UiText.Plural -> pluralStringResource(resId, count, *composeArgs(args))
}

@Composable
private fun composeArgs(args: List<Any>): Array<Any> =
    args.map { if (it is UiText) it.resolve() else it }.toTypedArray()
```

```kotlin
internal fun metricValueText(value: String, unit: String?, unitColor: Color): AnnotatedString = buildAnnotatedString {

@Immutable
data class StatCell(
    val label: String,
    val value: String,
    val unit: String? = null,
    val valueColor: Color = Color.Unspecified,
)

internal data class CellBorders(val end: Boolean, val bottom: Boolean)

/** Hairlines like mockup .grid3: right border except the last column/cell, bottom border except the last row. */
internal fun gridCellBorders(index: Int, count: Int, columns: Int): CellBorders {
    require(columns > 0) { "columns must be > 0" }
    require(index in 0 until count) { "index out of range" }
    val lastRowStart = ((count - 1) / columns) * columns
    return CellBorders(
        end = index % columns < columns - 1 && index < count - 1,
        bottom = index < lastRowStart,
    )
}

/** One card, [columns] equal cells per row, hairline separators (hr Detail 3×3 grid, spec §7.2). */
@Composable
fun StatGrid(cells: List<StatCell>, modifier: Modifier = Modifier, columns: Int = 3) {
    require(columns > 0) { "columns must be > 0" }
    val line = MaterialTheme.colorScheme.outline
    val shape = MaterialTheme.shapes.medium
    Column(
        modifier = modifier
            .fillMaxWidth()
            .clip(shape)
            .background(MaterialTheme.colorScheme.surface)
            .border(1.dp, line, shape),
    ) {
        cells.chunked(columns).forEachIndexed { row, rowCells ->
            Row(Modifier.fillMaxWidth().height(IntrinsicSize.Min)) {
                rowCells.forEachIndexed { column, cell ->
                    val borders = gridCellBorders(row * columns + column, cells.size, columns)
                    Metric(
                        label = cell.label,
                        value = cell.value,
                        unit = cell.unit,
                        valueColor = cell.valueColor,
                        modifier = Modifier
                            .weight(1f)
                            .fillMaxHeight()
                            .hairlines(borders, line)
                            .padding(vertical = AppTheme.spacing.md, horizontal = AppTheme.spacing.xs),
                    )
                }
                repeat(columns - rowCells.size) { Spacer(Modifier.weight(1f)) }
            }
        }
    }
}

private fun Modifier.hairlines(borders: CellBorders, color: Color): Modifier = drawBehind {
    val stroke = 1.dp.toPx()
    if (borders.end) {
        val x = if (layoutDirection == LayoutDirection.Ltr) size.width - stroke / 2 else stroke / 2
        drawLine(color, Offset(x, 0f), Offset(x, size.height), stroke)
    }
    if (borders.bottom) {
        val y = size.height - stroke / 2
        drawLine(color, Offset(0f, y), Offset(size.width, y), stroke)
    }
}
```

**Mockup mapping.**

- `.tabs`: hairline outline at the bottom, 10px .14em uppercase in textDisabled. The selected tab has white text and a 2dp inset accent underline.
- The selected style (accent border + tint, onSurface text) also addresses the HANDOVER "Z5 chip contrast" follow-up wherever an app adopts SegmentedControl for zones.

```kotlin
    selectedIndex: Int,

    modifier: Modifier = Modifier,
    enabled: Boolean = true,
```

```kotlin
import androidx.compose.ui.graphics.Color
```

```kotlin
import kotlinx.coroutines.launch

/** Fraction of the row width the card must travel before a release deletes. */
internal const val SwipeDeleteFraction = 0.4f

/** Alpha of the error tint behind a swiped row. */
internal const val SwipeTintAlpha = 0.18f

internal fun swipeDeleteThreshold(totalDistance: Float): Float = totalDistance * SwipeDeleteFraction

internal fun shouldRequestDelete(value: SwipeToDismissBoxValue, enabled: Boolean): Boolean =
    enabled && value == SwipeToDismissBoxValue.EndToStart

internal fun showDeleteBackground(direction: SwipeToDismissBoxValue): Boolean =
    direction == SwipeToDismissBoxValue.EndToStart

internal fun swipeDeleteContainer(error: Color): Color = error.copy(alpha = SwipeTintAlpha)

/**
 * Swipe a row towards the start to delete it. [onDelete] fires once per completed swipe; the row then
 * animates back, so a confirm dialog or undo snackbar finds it in place.
 */
@Composable
fun SwipeToDelete(
    onDelete: () -> Unit,
    modifier: Modifier = Modifier,
    enabled: Boolean = true,
    content: @Composable RowScope.() -> Unit,
) {
    val state = rememberSwipeToDismissBoxState(positionalThreshold = ::swipeDeleteThreshold)
    val currentOnDelete by rememberUpdatedState(onDelete)
    val currentEnabled by rememberUpdatedState(enabled)
    val scope = rememberCoroutineScope()
    // Stable identity: SwipeToDismissBox keys its LaunchedEffect on onDismiss (Review Focus 2).
    val handleDismiss: (SwipeToDismissBoxValue) -> Unit = remember(state, scope) {
        { value ->
            scope.launch { state.reset() }
            if (shouldRequestDelete(value, currentEnabled)) currentOnDelete()
        }
    }
    val error = MaterialTheme.colorScheme.error
    val shape = MaterialTheme.shapes.medium
    SwipeToDismissBox(
        state = state,
        modifier = modifier,
        enableDismissFromStartToEnd = false,
        enableDismissFromEndToStart = true,
        gesturesEnabled = enabled,
        onDismiss = handleDismiss,
        backgroundContent = {
            if (showDeleteBackground(state.dismissDirection)) {
                Box(
                    modifier = Modifier
                        .fillMaxSize()
                        .clip(shape)
                        .background(swipeDeleteContainer(error))
                        .padding(horizontal = AppTheme.spacing.lg),
                    contentAlignment = Alignment.CenterEnd,
                ) {
                    Icon(
                        imageVector = Icons.Outlined.Delete,
                        contentDescription = stringResource(R.string.shared_action_delete),
                        tint = error,
                    )
                }
            }
        },
        content = content,
    )
}
```

```kotlin
@Composable
fun TextInputDialog(

    initialValue: String,
    onConfirm: (String) -> Unit,
    onDismiss: () -> Unit,
    modifier: Modifier = Modifier,
    label: String? = null,
    placeholder: String? = null,

    allowEmpty: Boolean = false,
    confirmText: String = stringResource(R.string.shared_action_save),
    dismissText: String = stringResource(R.string.shared_action_cancel),

                keyboardActions = KeyboardActions(onDone = { confirm() }),
            )
        },
        actions = {
            PrimaryButton(text = confirmText, onClick = confirm, modifier = Modifier.fillMaxWidth(), enabled = canConfirm)
            TextActionButton(text = dismissText, onClick = onDismiss, modifier = Modifier.fillMaxWidth())

    options: List<T>,
    selected: T?,

    onConfirm: (T) -> Unit,
    onDismiss: () -> Unit,
    modifier: Modifier = Modifier,
    confirmText: String = stringResource(R.string.shared_action_confirm),
    dismissText: String = stringResource(R.string.shared_action_cancel),
) {
    var choice by remember(options, selected) { mutableStateOf(selected) }
    AppDialog(
        onDismissRequest = onDismiss,
        title = title,
        modifier = modifier,
        content = {
            Column(Modifier.selectableGroup().verticalScroll(rememberScrollState())) {

                    val isSelected = option == choice
```

```kotlin
package com.kevin.shared.ui.input

import java.math.BigDecimal
import java.math.RoundingMode

/** Result of parsing a numeric text field. */
sealed interface NumberInput {
    data object Empty : NumberInput
    data object Invalid : NumberInput
    data class OutOfRange(val value: Double) : NumberInput
    data class Valid(val value: Double) : NumberInput
}

private val IntegerPattern = Regex("[+-]?\\d+")
private val DecimalPattern = Regex("[+-]?(\\d+(\\.\\d*)?|\\.\\d+)")

/**
 * Parses user input: blank -> Empty, ',' accepted as decimal separator, [decimals] == 0 accepts integers only.
 * [range] is checked on the raw value before rounding HALF_UP to [decimals].
 */
fun parseNumberInput(text: String, range: ClosedFloatingPointRange<Double>? = null, decimals: Int): NumberInput {
    require(decimals >= 0) { "decimals must be >= 0" }
    val normalized = text.trim().replace(',', '.')
    if (normalized.isEmpty()) return NumberInput.Empty
    val pattern = if (decimals == 0) IntegerPattern else DecimalPattern
    if (!pattern.matches(normalized)) return NumberInput.Invalid
    val raw = normalized.toDoubleOrNull()?.takeIf { it.isFinite() } ?: return NumberInput.Invalid
    if (range != null && raw !in range) return NumberInput.OutOfRange(raw)
    return NumberInput.Valid(BigDecimal.valueOf(raw).setScale(decimals, RoundingMode.HALF_UP).toDouble())
}

/** True when the field should show its error: Invalid or OutOfRange. Empty is not an error. */
val NumberInput.isError: Boolean
    get() = this is NumberInput.Invalid || this is NumberInput.OutOfRange

fun NumberInput.valueOrNull(): Double? = (this as? NumberInput.Valid)?.value
```

## Open questions and risks (for the main session)

1. **Working tree.** Choose between the main checkout and a worktree before Task 1 (Global Constraints). App QA (Task 18) blocks until the main checkout can be switched to `ds-basics`.
2. **Visual changes reach both apps immediately** through `includeBuild`: `StatItem`, the lib dialogs, session/trash/BLE rows and `TargetZoneDialog` (Tasks 7 and 14–16). If that is unwanted before SP5/SP6, split Tasks 14–16 into a follow-up branch. Tasks 1–13 are additive.

6. **`ProgressDialog` name.** It shares a simple name with the deprecated `android.app.ProgressDialog`. Watch IDE auto-imports in SP5/SP6.
7. **`shared_session_duration`** becomes unused inside the lib. It is kept as a public resource; delete it after SP5/SP6 confirm no app uses it.
